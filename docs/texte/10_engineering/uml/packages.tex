% Version: 0.2.0 (2026-10-12) -- package signal (IQ simulation, FX correlator)

%   0.1.0: first draft, based on the approved repository structure
\begin{tikzpicture}[node distance=1cm]
  % ---- user interface layer ------------------------------------------
  \umlpkg{nb}{0,0}{notebooks}
  \umlpkg{widgets}{-2.6,-2.3}{widgets}
  \umlpkg{plotting}{2.6,-2.3}{plotting}

  % ---- simulation core (independent of the user interface) -----------
  \umlpkg{sky}{-4.6,-4.9}{sky}
  \umlpkg{instrument}{-1.5,-4.9}{instrument}
  \umlpkg{signal}{1.6,-4.9}{signal}
  \umlpkg{observation}{-3.05,-7.1}{observation}
  \umlpkg{imaging}{0.6,-7.1}{imaging}
  \umlpkg{deconvolution}{5.0,-7.1}{deconvolution}
  \umlframe[fill=gray!8]{core}{(sky)(instrument)(signal)(observation)(imaging)(deconvolution)}{core (UI-independent)}

  % ---- support -------------------------------------------------------
  \umlpkg{visdata}{0.0,-10.0}{visdata}
  \umlpkg{io}{-3.6,-10.0}{io}
  \umlpkg{constants}{5.0,-10.0}{constants}
  \umlframe{pkg}{(widgets)(plotting)(core)(io)(visdata)(constants)}{interferosim}

  % ---- configuration files (artefact, outside the package) -----------
  \node[umlpkgbox,fill=yellow!12,font=\scriptsize\sffamily,minimum width=2.2cm] (cfg)
        at (-9.0,-6.0) {{\tiny$\ll$artifact$\gg$}\\\textbf{configs/*}};

  % ---- dependencies --------------------------------------------------
  \draw[depend] (nb) -- node[stereo,left=2pt]{$\ll$use$\gg$} (widgets);
  \draw[depend] (nb) -- (plotting);
  \draw[depend] (widgets) -- node[stereo,above]{$\ll$use$\gg$} (plotting);
  \draw[depend] (widgets.south west) -- (widgets.south west |- core.north)
        node[stereo,pos=0.5,left=2pt]{$\ll$use$\gg$};
  \draw[depend] (plotting.south east) -- (plotting.south east |- core.north)
        node[stereo,pos=0.5,right=2pt]{$\ll$use$\gg$};

  \draw[depend] (observation) -- (sky);
  \draw[depend] (observation) -- (instrument);
  \draw[depend] (signal) -- node[stereo,above]{$\ll$use$\gg$} (instrument);
  \draw[depend] (signal.south west) to[out=200,in=-25] (sky.south east);
  \draw[depend] (signal) -- node[stereo,above,sloped]{geometry} (observation);
  \draw[depend] (imaging) -- (visdata);
  \draw[depend] (deconvolution) -- (imaging);
  \draw[depend] (core.west |- cfg) -- node[stereo,above]{$\ll$read$\gg$} (cfg);
  \draw[depend] (observation) -- node[stereo,left]{$\ll$create$\gg$} (visdata);
  \draw[depend] (signal.east) -- (3.3,-4.9) -- node[stereo,right,pos=0.3]{$\ll$create$\gg$} (3.3,-9.85) -- ($(visdata.east)+(0,0.15)$);
  \draw[depend] (io) -- node[stereo,below]{$\ll$read/write$\gg$} (visdata);
  \draw[depend] (core.south -| constants) -- node[stereo,right=2pt]{$\ll$use$\gg$} (constants);
\end{tikzpicture}
